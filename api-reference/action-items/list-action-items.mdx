---
title: 'List Action Items'
api: 'GET /api/v1/action-items'
description: 'List action items for your enterprise with optional filters'
---

## Query Parameters

<ParamField query="interaction_id" type="string">
  Filter by interaction UUID — returns all action items linked to a specific call, meeting, or thread. Also accepts `interactionId`.
</ParamField>

<ParamField query="owner_user_id" type="string">
  Filter by the UUID of the user who owns the action item
</ParamField>

<ParamField query="stage_id" type="string">
  Filter by pipeline stage UUID.
</ParamField>

<ParamField query="priority" type="string">
  Filter by priority. One of: `low`, `medium`, `high`, `blocker`
</ParamField>

<ParamField query="meeting_id" type="string">
  Filter by meeting UUID
</ParamField>

<ParamField query="pipeline_id" type="string">
  Filter by pipeline UUID
</ParamField>

<ParamField query="kind" type="string">
  Filter by action item kind
</ParamField>

<ParamField query="company_id" type="string">
  Filter by company UUID
</ParamField>

<ParamField query="unassigned" type="boolean">
  When `true`, return only action items with no owner
</ParamField>

<ParamField query="page" type="integer" default="1">
  Page number for pagination
</ParamField>

<ParamField query="limit" type="integer" default="20">
  Number of results per page
</ParamField>

<ParamField query="sort_by" type="string">
  Field to sort by
</ParamField>

<ParamField query="sort_order" type="string">
  Sort order. One of: `asc`, `desc`
</ParamField>
