## Overview

Returns a paginated list of call records for your enterprise with optional filtering by status, direction, company, account, contact, date range, and more.

## Query Parameters

<ParamField query="accountId" type="string">
  Only calls linked to this account (UUID). Comma-separate or repeat for several. Also accepts `account_id`.
</ParamField>

<ParamField query="accountExternalId" type="string">
  Only calls linked to the account with this external ID (your own ID, matched case-insensitively). Also accepts `externalAccountId`.
</ParamField>

<ParamField query="companyId" type="string">
  Only calls linked to this company (UUID). Comma-separate or repeat for several. Also accepts `company_id`.
</ParamField>

<ParamField query="companyExternalId" type="string">
  Only calls linked to the company with this external ID (matched case-insensitively). Also accepts `externalCompanyId`.
</ParamField>

<ParamField query="page" type="integer" default="1">
  Page number for pagination.
</ParamField>

<ParamField query="offset" type="integer">
  Number of results to skip. Takes precedence over `page`.
</ParamField>

<ParamField query="limit" type="integer" default="20">
  Number of results per page. Maximum is 100.
</ParamField>

<ParamField query="status" type="string">
  Filter by call status. One of: `completed`, `missed`, `voicemail`, `failed`, `busy`, `cancelled`, `unknown`
</ParamField>

<ParamField query="direction" type="string">
  Filter by direction. One of: `inbound`, `outbound`, `unknown`
</ParamField>

<ParamField query="startDate" type="string">
  Calls starting on or after this date or timestamp (ISO 8601). Also accepts `start_date`.
</ParamField>

<ParamField query="endDate" type="string">
  Calls starting on or before this date or timestamp (ISO 8601). A date without a time includes that whole day. Also accepts `end_date`.
</ParamField>

<ParamField query="search" type="string">
  Search by call title (case-insensitive partial match). Also accepts `q`.
</ParamField>

<ParamField query="provider" type="string">
  Filter by integration provider (e.g. `aircall`, `twilio`, `nooks`)
</ParamField>

<ParamField query="providerKey" type="string">
  Filter by integration provider key
</ParamField>

<ParamField query="participant_only" type="boolean">
  When `true`, return only calls where the requesting user is a participant
</ParamField>

<ParamField query="date_range" type="string">
  Relative date range filter. Examples: `7d`, `30d`
</ParamField>

<ParamField query="org_unit_id" type="string">
  Filter by org unit UUID
</ParamField>

<ParamField query="user_id" type="string">
  Filter by user UUID
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Whether the operation was successful
</ResponseField>

<ResponseField name="data" type="array">
  Calls, most recent first. Each call also has:

  <Expandable title="Company and account links">
    <ResponseField name="companyIds" type="string[]">UUIDs of the companies the call is linked to</ResponseField>
    <ResponseField name="accountIds" type="string[]">UUIDs of the accounts the call is linked to</ResponseField>
    <ResponseField name="companies" type="array">`{ id, name, externalId }` for each linked company</ResponseField>
    <ResponseField name="accounts" type="array">`{ id, name, externalId, companyId }` for each linked account</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="meta" type="object">
  Pagination metadata
</ResponseField>
