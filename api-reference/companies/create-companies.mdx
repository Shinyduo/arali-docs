---
title: 'Create Companies'
api: 'POST /api/v1/companies'
description: 'Batch create or update company records'
---

## Request Body

<ParamField body="providerKey" type="string">
  Accepted for backward compatibility only. It is not used for company lookup.
</ParamField>

<ParamField body="companies" type="array" required>
  Array of company objects to create or update
  
  <Expandable title="Company Object">
    <ParamField body="name" type="string" required>
      Company display name
    </ParamField>
    
    <ParamField body="domain" type="string">
      Company website domain
    </ParamField>
    
    <ParamField body="externalCompanyId" type="string">
      **External Company ID**. Your internal unique ID for this company (e.g. `ORG_123`, HubSpot Company ID). 
      Stored on `companies.external_company_id` and used for idempotent upserts. If a company with this `externalCompanyId` already exists, it will be updated instead of created.
    </ParamField>
    
    <ParamField body="stageKey" type="string" default="lead">
      Company stage key. Options: `lead`, `prospect`, `customer`, `expansion`, `churned`, `lost`, `partner`, `other`
    </ParamField>
    
    <ParamField body="ownerUserId" type="string">
      Owner user UUID
    </ParamField>

    <ParamField body="arr" type="number">
      Annual Recurring Revenue
    </ParamField>

    <ParamField body="currency" type="string" default="USD">
      Currency code (e.g. `USD`, `EUR`)
    </ParamField>
    
    <ParamField body="attributes" type="object">
      Custom key-value attributes
    </ParamField>
    <ParamField body="properties" type="object">
      Custom field values (maps to field_values table)
    </ParamField>

    <ParamField body="createdAt" type="string">
      ISO 8601 creation timestamp. If provided, this value is used as the company's `created_at` instead of the current time. Applied on both insert and upsert.
    </ParamField>

    <ParamField body="key_roles" type="array">
      Array of key role assignments for this company.
      
      <Expandable title="Key Role Object">
        <ParamField body="roleKey" type="string" required>Role key (e.g. `csm`, `ae`)</ParamField>
        <ParamField body="userEmail" type="string">Email of the assigned user</ParamField>
        <ParamField body="userId" type="string">UUID of the assigned user</ParamField>
      </Expandable>
    </ParamField>
  </Expandable>
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Whether the operation was successful
</ResponseField>

<ResponseField name="summary" type="object">
  <Expandable title="Summary Object">
    <ResponseField name="total" type="integer">
      Total number of companies processed
    </ResponseField>
    <ResponseField name="success" type="integer">
      Number of companies successfully created/updated
    </ResponseField>
    <ResponseField name="failed" type="integer">
      Number of companies that failed
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="results" type="array">
  Array of result objects for each company
</ResponseField>
