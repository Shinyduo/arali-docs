---
title: 'Get Company'
api: 'GET /api/v1/companies/{id}'
description: 'Get a single enriched company by ID'
---

## Path Parameters

<ParamField path="id" type="string" required>
  Company UUID
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Whether the operation was successful
</ResponseField>

<ResponseField name="data" type="object">
  Enriched company object with related contacts, deals, and accounts.

  <Expandable title="Company fields">
    <ResponseField name="id" type="string">Company UUID</ResponseField>
    <ResponseField name="enterpriseId" type="string">Enterprise UUID</ResponseField>
    <ResponseField name="name" type="string">Company display name</ResponseField>
    <ResponseField name="domain" type="string | null">Company website domain</ResponseField>
    <ResponseField name="ownerUserId" type="string | null">Owner user UUID</ResponseField>
    <ResponseField name="healthScore" type="integer | null">Health score (0-100)</ResponseField>
    <ResponseField name="healthScoreComputedAt" type="string | null">ISO 8601 datetime</ResponseField>
    <ResponseField name="healthScoreBreakdown" type="object | null">Breakdown object</ResponseField>
    <ResponseField name="ARR" type="integer | null">Annual Recurring Revenue</ResponseField>
    <ResponseField name="currency" type="string">Currency code (default: USD)</ResponseField>
    <ResponseField name="stageDefinitionId" type="string | null">Stage definition UUID</ResponseField>
    <ResponseField name="attributes" type="object">Custom key-value attributes</ResponseField>
    <ResponseField name="externalCompanyId" type="string | null">Secondary External ID</ResponseField>
    <ResponseField name="createdAt" type="string">Creation timestamp</ResponseField>
    <ResponseField name="updatedAt" type="string">Last updated timestamp</ResponseField>

    <ResponseField name="stage" type="object | null">
      Resolved stage definition.
      <Expandable title="Stage properties">
        <ResponseField name="id" type="string">Stage UUID</ResponseField>
        <ResponseField name="key" type="string">Stage key</ResponseField>
        <ResponseField name="name" type="string">Stage name</ResponseField>
        <ResponseField name="bucket" type="string">Stage bucket</ResponseField>
        <ResponseField name="scope" type="string">Stage scope</ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="fieldValues" type="array">
      Custom field values.
      <Expandable title="Field value properties">
        <ResponseField name="fieldKey" type="string">Field key</ResponseField>
        <ResponseField name="fieldName" type="string">Field name</ResponseField>
        <ResponseField name="fieldType" type="string">Field type</ResponseField>
        <ResponseField name="value" type="any">Value</ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="contacts" type="array">
      Related contacts.
      <Expandable title="Contact properties">
        <ResponseField name="id" type="string">Contact UUID</ResponseField>
        <ResponseField name="fullName" type="string">Full name</ResponseField>
        <ResponseField name="title" type="string">Job title</ResponseField>
        <ResponseField name="emails" type="array">[{ id, email, isPrimary }]</ResponseField>
        <ResponseField name="phones" type="array">[{ id, phone, isPrimary }]</ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="deals" type="array">
      Related deals.
      <Expandable title="Deal properties">
        <ResponseField name="id" type="string">Deal UUID</ResponseField>
        <ResponseField name="name" type="string">Deal name</ResponseField>
        <ResponseField name="amountCents" type="number | null">Amount in cents</ResponseField>
        <ResponseField name="currency" type="string">Currency</ResponseField>
        <ResponseField name="stage" type="object">Pipeline stage</ResponseField>
        <ResponseField name="pipeline" type="object">Pipeline</ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="accounts" type="array">
      Related accounts.
      <Expandable title="Account properties">
        <ResponseField name="id" type="string">Account UUID</ResponseField>
        <ResponseField name="name" type="string">Account name</ResponseField>
        <ResponseField name="domain" type="string">Domain</ResponseField>
        <ResponseField name="accountType" type="string">Account type</ResponseField>
        <ResponseField name="stage" type="object">Stage definition</ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
