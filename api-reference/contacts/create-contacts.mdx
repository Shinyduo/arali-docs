---
title: 'Create Contacts'
api: 'POST /api/v1/contacts'
description: 'Batch create or update contact records'
---

## Request Body

<ParamField body="providerKey" type="string" required>
  Integration provider key (e.g. `hubspot`, `salesforce`)
</ParamField>

<ParamField body="contacts" type="array" required>
  Array of contact objects to create or update
  
  <Expandable title="Contact Object">
    <ParamField body="externalContactId" type="string" required>
      External unique ID from integration source
    </ParamField>

    <ParamField body="fullName" type="string">
      Contact's full name
    </ParamField>

    <ParamField body="title" type="string">
      Job title
    </ParamField>

    <ParamField body="ownerUserId" type="string">
      Owner user UUID
    </ParamField>

    <ParamField body="emails" type="array">
      Array of email objects: `{ email: string, isPrimary?: boolean, label?: string }`
    </ParamField>

    <ParamField body="phones" type="array">
      Array of phone objects: `{ phone: string, isPrimary?: boolean, label?: string }`
    </ParamField>

    <ParamField body="attributes" type="object">
      Custom key-value attributes
    </ParamField>

    <ParamField body="stageKey" type="string">
      Contact stage key
    </ParamField>

    <ParamField body="stageDefinitionId" type="string">
      Stage definition UUID
    </ParamField>

    <ParamField body="properties" type="object">
      Custom field values
    </ParamField>

    <ParamField body="createdAt" type="string">
      ISO 8601 creation timestamp override
    </ParamField>

    <ParamField body="key_roles" type="array">
      Array of key role assignments.
      
      <Expandable title="Key Role Object">
        <ParamField body="roleKey" type="string" required>Role key</ParamField>
        <ParamField body="userEmail" type="string">Email of the assigned user</ParamField>
        <ParamField body="userId" type="string">UUID of the assigned user</ParamField>
      </Expandable>
    </ParamField>
  </Expandable>
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Whether the operation was successful
</ResponseField>

<ResponseField name="summary" type="object">
  <Expandable title="Summary Object">
    <ResponseField name="total" type="integer">Total contacts processed</ResponseField>
    <ResponseField name="success" type="integer">Successfully created/updated</ResponseField>
    <ResponseField name="failed" type="integer">Failed</ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="results" type="array">
  Array of result objects
</ResponseField>

## Example Request

```bash
curl -X POST https://api.arali.ai/api/v1/contacts \
  -H "Authorization: Api-Key YOUR_STATIC_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "providerKey": "hubspot",
    "contacts": [
      {
        "externalContactId": "contact_12345",
        "fullName": "John Doe",
        "title": "VP of Engineering",
        "emails": [
          { "email": "john.doe@acme.com", "isPrimary": true, "label": "Work" }
        ],
        "phones": [
          { "phone": "+1-555-123-4567", "isPrimary": true, "label": "Mobile" }
        ],
        "attributes": {
          "linkedin": "https://linkedin.com/in/johndoe"
        }
      }
    ]
  }'
```
