---
title: 'List Contacts'
api: 'GET /api/v1/contacts'
description: 'Get all contacts for your enterprise with optional filtering'
---

## Query Parameters

<ParamField query="page" type="integer" default="1">
  Page number to return. Must be greater than 0.
</ParamField>

<ParamField query="limit" type="integer" default="50">
  Number of results per page. Maximum is 200.
</ParamField>

<ParamField query="search" type="string">
  Search contacts by name, title, external contact ID, email, phone, or normalized phone.
</ParamField>

<ParamField query="q" type="string">
  Alias for `search`.
</ParamField>

<ParamField query="phone" type="string">
  Look up contacts by phone number using digits-only normalization.
</ParamField>

<ParamField query="email" type="string">
  Look up contacts by exact email address.
</ParamField>

<ParamField query="company_id" type="string">
  Filter by company UUID.
</ParamField>

<ParamField query="stage" type="string">
  Filter by contact stage key or stage definition UUID.
</ParamField>

<ParamField query="key_role_user_id" type="string">
  Filter by assigned key-role user UUID. Supports comma-separated values.
</ParamField>

<ParamField query="key_role_key" type="string">
  Filter by key role key. Supports comma-separated values.
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Whether the operation was successful
</ResponseField>

<ResponseField name="data" type="array">
  Array of lightweight contact objects. Use `GET /api/v1/contacts/{id}` for full enriched contact details.
</ResponseField>

<ResponseField name="meta" type="object">
  <Expandable title="Meta Object">
    <ResponseField name="total" type="integer">Total number of matching contacts</ResponseField>
    <ResponseField name="count" type="integer">Number of contacts returned in this response</ResponseField>
    <ResponseField name="page" type="integer">Current page number</ResponseField>
    <ResponseField name="limit" type="integer">Results per page</ResponseField>
    <ResponseField name="totalPages" type="integer">Total number of pages for the current filters</ResponseField>
    <ResponseField name="hasMore" type="boolean">Whether another page is available</ResponseField>
    <ResponseField name="enterpriseId" type="string">Enterprise UUID used to scope the request</ResponseField>
  </Expandable>
</ResponseField>
