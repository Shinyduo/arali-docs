---
title: 'Create Pipelines'
api: 'POST /api/v1/pipelines'
description: 'Batch create or update pipeline records'
---

## Request Body

<ParamField body="pipelines" type="array" required>
  Array of pipeline objects to create or update
  
  <Expandable title="Pipeline Object">
    <ParamField body="id" type="string">
      Pipeline UUID. Provide to update an existing pipeline.
    </ParamField>

    <ParamField body="name" type="string" required>
      Pipeline display name
    </ParamField>

    <ParamField body="entityType" type="string" default="action_item">
      Entity type this pipeline applies to. One of: `deal`, `action_item`, `ticket`
    </ParamField>

    <ParamField body="orgUnitId" type="string">
      Org unit ID (required only for `action_item` pipelines)
    </ParamField>

    <ParamField body="description" type="string">
      Pipeline description
    </ParamField>

    <ParamField body="isDefault" type="boolean" default="false">
      Whether this is the default pipeline for its entity type
    </ParamField>

    <ParamField body="createdByUserId" type="string">
      User UUID who created this pipeline
    </ParamField>
  </Expandable>
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Whether the operation was successful
</ResponseField>

<ResponseField name="summary" type="object">
  Summary of the batch operation

  <Expandable title="Summary Object">
    <ResponseField name="total" type="number">
      Total number of pipelines submitted
    </ResponseField>

    <ResponseField name="success" type="number">
      Number of pipelines successfully created/updated
    </ResponseField>

    <ResponseField name="failed" type="number">
      Number of pipelines that failed
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="results" type="array">
  Array of individual operation results

  <Expandable title="Result Object">
    <ResponseField name="name" type="string">
      Pipeline name
    </ResponseField>

    <ResponseField name="pipelineId" type="string">
      UUID of the created/updated pipeline (present on success)
    </ResponseField>

    <ResponseField name="status" type="string">
      Result status. One of: `created`, `updated`, `error`
    </ResponseField>

    <ResponseField name="error" type="string">
      Error message if the operation failed
    </ResponseField>
  </Expandable>
</ResponseField>
