---
title: 'Query Product Events'
api: 'GET /api/v1/product-events'
description: 'Query product event records'
---

## Query Parameters

<ParamField query="limit" type="integer" default="100">
  Number of results per page. Maximum is 500.
</ParamField>

<ParamField query="offset" type="integer" default="0">
  Number of results to skip
</ParamField>

<ParamField query="eventName" type="string">
  Filter by event name
</ParamField>

<ParamField query="eventKey" type="string">
  Filter by event key
</ParamField>

<ParamField query="companyId" type="string">
  Filter by company UUID
</ParamField>

<ParamField query="accountId" type="string">
  Filter by account UUID
</ParamField>

<ParamField query="productId" type="string">
  Filter by product UUID
</ParamField>

<ParamField query="startAt" type="string">
  Filter events after this timestamp (ISO 8601)
</ParamField>

<ParamField query="endAt" type="string">
  Filter events before this timestamp (ISO 8601)
</ParamField>

## Response

<ResponseField name="success" type="boolean">
  Whether the operation was successful
</ResponseField>

<ResponseField name="data" type="array">
  Array of product event objects
</ResponseField>
